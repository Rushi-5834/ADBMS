CREATE DATABASE theatre_db;
CREATE DATABASE payment_db;

USE theatre_db;

CREATE TABLE seats (
    seat_no VARCHAR(10) PRIMARY KEY,
    status VARCHAR(20)
);

INSERT INTO seats VALUES
('S12', 'Available');

USE payment_db;

CREATE TABLE payments (
    payment_id INT PRIMARY KEY,
    seat_no VARCHAR(10),
    amount INT,
    status VARCHAR(20)
);

-- Start transaction
START TRANSACTION;

-- Book the seat
UPDATE theatre_db.seats
SET status = 'Booked'
WHERE seat_no = 'S12'
AND status = 'Available';

-- Insert payment details
INSERT INTO payment_db.payments
VALUES (1, 'S12', 500, 'Success');

-- Save changes
COMMIT;

-- Check results
SELECT * FROM theatre_db.seats;
SELECT * FROM payment_db.payments;
