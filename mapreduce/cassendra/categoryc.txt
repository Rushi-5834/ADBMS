-- Step 1: Create keyspace
CREATE KEYSPACE college
WITH replication = {
    'class': 'SimpleStrategy',
    'replication_factor': 1
};

-- Step 2: Select keyspace
USE college;

-- Step 3: Create table
CREATE TABLE products (
    product_id INT PRIMARY KEY,
    product_name TEXT,
    category TEXT,
    price INT
);
