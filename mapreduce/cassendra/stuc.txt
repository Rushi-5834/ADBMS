-- Step 1: Create keyspace
CREATE KEYSPACE college
WITH replication = {
    'class': 'SimpleStrategy',
    'replication_factor': 1
};

-- Step 2: Select keyspace
USE college;

-- Step 3: Create table
CREATE TABLE students (
    roll_no INT PRIMARY KEY,
    name TEXT,
    department TEXT,
    marks INT
);
